from pathlib import Path
import json,numpy as np
folder=Path('/tmp/soccer-v7-waist36');data=json.loads((folder/'decoded-regression.json').read_text());fixtures=json.loads((folder/'regression-fixtures.json').read_text())['fixtures'];M=lambda a:np.array(a,dtype=float).reshape(4,4).T
cache={}
for row in data:
 part=next(p for p in row['parts'] if p['surface']=='Cloth');v=np.array(part['position']).reshape(-1,3);j=np.array(part['joints']).reshape(-1,4);w=np.array(part['weights']).reshape(-1,4);w/=w.sum(1)[:,None];full=np.zeros((len(v),18));
 for k in range(4):np.add.at(full,(np.arange(len(v)),j[:,k]),w[:,k])
 cache[(row['variant'],row['lod'])]=(np.c_[v,np.ones(len(v))],full,row['joints'])
records=[]
for f in fixtures:
 mesh=next(m for m in f['meshes'] if m['material']==['Cloth']);left=M(mesh['matrixWorld'])@M(mesh['bindMatrixInverse']);right=M(mesh['bindMatrix']);mins={}
 for key,(v,w,names) in cache.items():
  y=np.zeros(len(v))
  for k,name in enumerate(names):
   if not np.any(w[:,k]):continue
   transform=left@M(f['bones'][name]['matrixWorld'])@M(mesh['boneInverses'][k])@right;y+=(v@transform[1])*w[:,k]
  mins[key]=float(y.min())
 for lod in [0,1]:
  b=mins[('baseline',lod)];c=mins[('candidate',lod)];records.append({'label':f['label'],'lod':lod,'baselineClothMinimumY':b,'candidateClothMinimumY':c,'deltaMinimumMeters':c-b,'baselineActualNearFullMeshMinimumY':f['minimumY'],'candidateFullMeshBound':min(c,f['minimumY']),'newFloorRegression':c<min(b,f['minimumY'])-.001})
# Bone poses and all Skin attributes/weights are identical, so every delivered
# Skin vertex is mathematically unchanged; that alone is not a cloth occlusion test.
summary={'cases':len(records),'newFloorRegressions':sum(r['newFloorRegression'] for r in records),'minimumCandidateClothY':min(r['candidateClothMinimumY'] for r in records),'worstClothMinimumDelta':min(r['deltaMinimumMeters'] for r in records),'limits':'Same fixed baseline Near skeletal poses applied to each actual decoded LOD. Near full-mesh minimum is a bound for unchanged non-Cloth only on near; far baseline full-mesh floor requires final runtime audit. This is not a new canonical floor PASS.'}
(folder/'floor-regression.json').write_text(json.dumps({'kind':'waist36-fixed-pose-Cloth-floor-regression','summary':summary,'records':records},indent=2)+'\n');print(summary)
for r in records:
 if r['newFloorRegression']:print('FAIL',r)
