M=lambda a:np.array(a,float).reshape(4,4).T

def prepared(d,surface):
 p=next(p for p in d['parts'] if p['surface']==surface);v=np.array(p['position']).reshape(-1,3);uv=np.array(p['uv']).reshape(-1,2);j=np.array(p['joints']).reshape(-1,4);w=np.array(p['weights']).reshape(-1,4);full=np.zeros((len(v),18))
 for k in range(4):full[np.arange(len(v)),j[:,k]]+=w[:,k]
 tri=np.array(p['indices']).reshape(-1,3)
 side=np.where(v[tri,0].mean(1)>=0,0,1)
 if surface=='Skin':
  leg=full[:,12:18].sum(1);keep=(leg[tri].mean(1)>.75)&(v[tri,1].max(1)>.61)&(v[tri,1].min(1)<.67)
  left=full[:,12:15].sum(1);right=full[:,15:18].sum(1);side=np.where(left[tri].mean(1)>=right[tri].mean(1),0,1)
 else:
  short=(uv[:,0]>.615)&(uv[:,0]<.995)&(uv[:,1]>.455)&(uv[:,1]<.645);keep=short[tri].all(1)
 return {'v':v,'full':full,'tri':tri[keep],'side':side[keep],'surface':surface}

def posed(p,d,f):
 v=p['v'];mesh=next(m for m in f['meshes'] if m['material']==[p['surface']]);vb=np.c_[v,np.ones(len(v))]@M(mesh['bindMatrix']).T;out=np.zeros_like(vb)
 for k,name in enumerate(d['joints']):out+=(vb@(M(f['bones'][name]['matrixWorld'])@M(mesh['boneInverses'][k])).T)*p['full'][:,k,None]
 world=out@(M(mesh['matrixWorld'])@M(mesh['bindMatrixInverse'])).T
 cam=np.eye(4);cam[:3,:3]=Rotation.from_quat(f['camera']['quaternion']).as_matrix();cam[:3,3]=f['camera']['position'];clip=world@(M(f['camera']['projectionMatrix'])@np.linalg.inv(cam)).T;ndc=clip[:,:3]/clip[:,3,None]
 return np.c_[(ndc[:,0]+1)*W/2,(1-ndc[:,1])*H/2,ndc[:,2]],1/clip[:,3]

def raster(screen,invw,tri,attr):
 depth=np.full((H,W),np.inf);ymap=np.full((H,W),np.nan);tid=np.full((H,W),-1,int)
 for i,t in enumerate(tri):
  q=screen[t];xmin=max(0,int(np.floor(q[:,0].min())));xmax=min(W-1,int(np.ceil(q[:,0].max())));ymin=max(0,int(np.floor(q[:,1].min())));ymax=min(H-1,int(np.ceil(q[:,1].max())))
  if xmax<xmin or ymax<ymin:continue
  a,b,c=q;den=(b[1]-c[1])*(a[0]-c[0])+(c[0]-b[0])*(a[1]-c[1])
  # Actual production Skin and Cloth use front faces only. Projection flips Y.
  if den>=-1e-12:continue
  yy,xx=np.mgrid[ymin:ymax+1,xmin:xmax+1];xx=xx+.5;yy=yy+.5;u=((b[1]-c[1])*(xx-c[0])+(c[0]-b[0])*(yy-c[1]))/den;v=((c[1]-a[1])*(xx-c[0])+(a[0]-c[0])*(yy-c[1]))/den;w=1-u-v;z=u*a[2]+v*b[2]+w*c[2];old=depth[ymin:ymax+1,xmin:xmax+1];hit=(u>=0)&(v>=0)&(w>=0)&(z<old)
  old[hit]=z[hit];tid[ymin:ymax+1,xmin:xmax+1][hit]=i
  av=attr[t]*invw[t];rest=(u*av[0]+v*av[1]+w*av[2])/(u*invw[t[0]]+v*invw[t[1]]+w*invw[t[2]]);ymap[ymin:ymax+1,xmin:xmax+1][hit]=rest[hit]
 return depth,ymap,tid
